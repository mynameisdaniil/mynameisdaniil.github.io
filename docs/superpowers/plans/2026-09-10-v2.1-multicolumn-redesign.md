# v2.1 Multicolumn Whitepaper Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the graph-of-cards UI with a white/black, masthead-first whitepaper whose sections flow across 1–4 CSS columns, keeping the existing typography engines.

**Architecture:** `content.yml` gains `meta.subtitle`/`meta.links` and `sections` (replacing `cards`, `default_card`, internal links, and edges). `main.js` renders a static masthead plus flow sections; `site.css` owns the multi-column layout and hairlines. All graph/card/router/theme code and d3 are deleted.

**Tech Stack:** Vanilla ES modules, CSS multi-column, marked, js-yaml, tex-linebreak + Hypher + protrusion, Tailwind preflight, `node:test`.

**Spec:** `docs/superpowers/specs/2026-09-10-v2.1-multicolumn-redesign-design.md`

## Global Constraints

- White background (`#fff`), black text and rules (`#000`) only. No dark mode, no grays, no toggle.
- Column counts: 1 (<640px), 2 (640–1023), 3 (1024–1439), 4 (≥1440). `column-rule: 1px solid #000`; `column-gap: 2.5rem`.
- Page container `max-width: 1600px`, centered, padding `clamp(1.5rem, 4vw, 4rem)`; masthead left-aligned.
- Fonts: `"GOST Type A"` for `h1`/`h2`, `"Latin Modern Roman"` body. No new dependencies.
- `meta.title`, `meta.subtitle` required; links are external/`mailto:` only (`label` + `url`).
- Sections keep `content.yml` order and the existing copy; facts stay 21 paragraphs.
- Typography pipeline unchanged (Knuth-Plass justification, hyphenation, protrusion), re-run debounced 150 ms on resize.
- Every task ends with a commit; no `node_modules` or `.superpowers` commits.

```js
const ID_RE = /^[a-z0-9-]+$/;

function parseLinks(rawLinks, where, errors) {
  const links = [];
  if (rawLinks === undefined) return links;
  if (!Array.isArray(rawLinks)) {
    errors.push(`${where} must be a list`);
    return links;
  }
  for (const [i, item] of rawLinks.entries()) {
    const path = `${where}[${i}]`;
    if (!item || typeof item !== "object" || Array.isArray(item)) {
      errors.push(`${path} must be a mapping`);
      continue;
    }
    if (typeof item.url !== "string" || item.url.length === 0) {
      errors.push(`${path} must have a url`);
      continue;
    }
    const label = typeof item.label === "string" && item.label ? item.label : item.url;
    links.push({ label, url: item.url });
  }
  return links;
}

export function buildModel(raw) {
  if (!raw || typeof raw !== "object" || Array.isArray(raw)) {
    return {
      meta: { title: "", subtitle: "", links: [] },
      sections: [],
      errors: ["content.yml: root must be a mapping"],
    };
  }

  const errors = [];
  const metaRaw = raw.meta && typeof raw.meta === "object" && !Array.isArray(raw.meta) ? raw.meta : {};
  const title = typeof metaRaw.title === "string" ? metaRaw.title : "";
  const subtitle = typeof metaRaw.subtitle === "string" ? metaRaw.subtitle : "";
  if (!title) errors.push("meta.title is required");
  if (!subtitle) errors.push("meta.subtitle is required");
  const metaLinks = parseLinks(metaRaw.links, "meta.links", errors);

  const sections = [];
  const seen = new Set();
  if (!Array.isArray(raw.sections)) {
    errors.push("sections must be a list");
  } else if (raw.sections.length === 0) {
    errors.push("sections must not be empty");
  }

  for (const [i, section] of (Array.isArray(raw.sections) ? raw.sections : []).entries()) {
    const where = `sections[${i}]`;
    if (!section || typeof section !== "object" || Array.isArray(section)) {
      errors.push(`${where} must be a mapping`);
      continue;
    }
    const id = typeof section.id === "string" ? section.id : "";
    const sectionTitle = typeof section.title === "string" ? section.title : "";
    if (!id) errors.push(`${where}.id is required`);
    else if (!ID_RE.test(id)) errors.push(`${where}.id must match [a-z0-9-]+`);
    else if (seen.has(id)) errors.push(`duplicate section id "${id}"`);
    if (!sectionTitle) errors.push(`${where}.title is required`);
    if (!id || seen.has(id)) continue;

    seen.add(id);
    sections.push({
      id,
      title: sectionTitle,
      body: typeof section.body === "string" ? section.body : "",
      lang: typeof section.lang === "string" ? section.lang : "",
      links: parseLinks(section.links, `${where}.links`, errors),
    });
  }

  return { meta: { title, subtitle, links: metaLinks }, sections, errors };
}

export function isValid(model) {
  return model.errors.length === 0;
}
```

import { applyProtrusion, clearProtrusion } from "./protrusion.js";

export function createTypography({ getHyphenator, typography = globalThis.typography }) {
  function applyTo(el, lang = "en-us") {
    const paragraphs = [...el.querySelectorAll("p")];
    if (paragraphs.length === 0) return;

    try {
      clearProtrusion(el);
      typography.justifyContent(paragraphs, getHyphenator(lang));
      applyProtrusion(el);
    } catch (err) {
      console.error("typography: justification failed", err);
    }
  }

  return { applyTo };
}

```js
import { loadContent, renderMarkdown, ContentError } from "./content.js";
import { createTypography } from "./typography.js";

const page = document.querySelector(".page");
const masthead = document.querySelector(".masthead");
const titleEl = document.getElementById("site-title");
const subtitleEl = document.getElementById("site-subtitle");
const linksEl = document.getElementById("site-links");
const paper = document.getElementById("paper");

const typography = createTypography({
  getHyphenator: (lang) => globalThis.typography.hyphenateFor(lang),
});

function createLink(link) {
  const a = document.createElement("a");
  a.textContent = link.label;
  a.href = link.url;
  if (!link.url.startsWith("mailto:")) {
    a.target = "_blank";
    a.rel = "noopener noreferrer";
  }
  return a;
}

function renderLinks(container, links) {
  container.replaceChildren(...links.map(createLink));
}

function renderSection(section) {
  const el = document.createElement("section");
  el.id = section.id;

  const heading = document.createElement("h2");
  heading.textContent = section.title;
  el.appendChild(heading);

  if (section.body) {
    const body = document.createElement("div");
    body.className = "section-body";
    body.innerHTML = renderMarkdown(section.body);
    el.appendChild(body);
  }

  if (section.links.length > 0) {
    const nav = document.createElement("nav");
    nav.className = "links-row section-links";
    nav.setAttribute("aria-label", `Links for ${section.title}`);
    renderLinks(nav, section.links);
    el.appendChild(nav);
  }

  return el;
}

let model = null;

function applyTypography() {
  if (!model) return;
  for (const section of model.sections) {
    const el = document.getElementById(section.id);
    if (el) typography.applyTo(el, section.lang || "en-us");
  }
}

function showError(details) {
  masthead.hidden = true;
  paper.replaceChildren();
  paper.classList.add("paper--error");

  const message = document.createElement("p");
  message.textContent = "Could not load content.";
  const list = document.createElement("ul");
  for (const detail of details) {
    const item = document.createElement("li");
    item.textContent = detail;
    list.appendChild(item);
  }
  paper.append(message, list);
}

async function boot() {
  try {
    model = await loadContent("content.yml");
    document.title = model.meta.title;
    titleEl.textContent = model.meta.title;
    subtitleEl.textContent = model.meta.subtitle;
    renderLinks(linksEl, model.meta.links);
    paper.replaceChildren(...model.sections.map(renderSection));
    await document.fonts.ready;
    applyTypography();
  } catch (err) {
    console.error(err);
    showError(err instanceof ContentError ? err.details : [String(err)]);
  }
}

let resizeTimer = 0;
window.addEventListener("resize", () => {
  clearTimeout(resizeTimer);
  resizeTimer = setTimeout(applyTypography, 150);
});

boot();
```
